# CodexTask 常用命令：按场景选择一组执行，不要从上到下全部运行。每条命令均为可直接复制执行的单行命令。

# === 场景：首次安装或升级 CLI；这是所有本机命令和自托管服务的基础。 ===
npm install -g codex-task@latest
codex-task --version
codex-task doctor

# === 场景：不再使用 CLI 时才卸载；不会删除 Codex 登录和普通项目文件。 ===
npm uninstall -g codex-task

# === 场景：把 Companion Skill 安装给其他 Agent；SkillTruck 安装 Skill，npm 安装上面的 CLI。 ===
npm install -g skilltruck
skilltruck install https://github.com/wangyendt/codex-task --global
codex-task skill path

# === 场景：第一次需要自启动脚本、但本机没有仓库；克隆后进入仓库根目录。 ===
git clone https://github.com/wangyendt/codex-task.git
cd codex-task

# === 场景：macOS 长期运行服务；在仓库根目录执行，安装 npm latest、立即启动，并在以后每次登录时自动启动。 ===
bash ./scripts/service/install.sh
curl -sS http://127.0.0.1:7777/healthz
cat "$HOME/Library/Application Support/codex-task/service/token"

# === macOS：仅在当前终端要运行 HTTP curl 示例时执行；不会影响服务，手机 App 也不读取这个环境变量。 ===
export CODEX_TASK_TOKEN="$(cat "$HOME/Library/Application Support/codex-task/service/token")"

# === macOS：查看服务状态。 ===
launchctl print "gui/$(id -u)/com.wangyendt.codex-task"

# === macOS：持续查看日志；按 Ctrl+C 退出日志查看，不会停止服务。 ===
tail -f "$HOME/Library/Application Support/codex-task/service/service.log"

# === macOS：配置变化或升级后重启服务。 ===
launchctl kickstart -k "gui/$(id -u)/com.wangyendt.codex-task"

# === macOS：不再需要自动启动服务时才卸载；保留全局 npm 包和 Codex 数据。 ===
bash ./scripts/service/uninstall.sh

# === 场景：Ubuntu/Linux 长期运行服务；在仓库根目录执行，安装 npm latest、立即启动，并默认随用户会话启动。 ===
bash ./scripts/service/install.sh
curl -sS http://127.0.0.1:7777/healthz
cat "${XDG_CONFIG_HOME:-$HOME/.config}/codex-task/service/token"

# === Ubuntu/Linux：仅在当前终端要运行 HTTP curl 示例时执行。 ===
export CODEX_TASK_TOKEN="$(cat "${XDG_CONFIG_HOME:-$HOME/.config}/codex-task/service/token")"

# === Ubuntu/Linux：查看服务状态。 ===
systemctl --user status codex-task.service

# === Ubuntu/Linux：持续查看日志；按 Ctrl+C 退出日志查看，不会停止服务。 ===
journalctl --user -u codex-task.service -f

# === Ubuntu/Linux：配置变化或升级后重启服务。 ===
systemctl --user restart codex-task.service

# === Ubuntu/Linux：只有需要“用户尚未登录也启动”时才执行。 ===
sudo loginctl enable-linger "$USER"

# === Ubuntu/Linux：不再需要自动启动服务时才卸载；保留全局 npm 包和 Codex 数据。 ===
bash ./scripts/service/uninstall.sh

# === 场景：Windows 长期运行服务；在仓库根目录的 PowerShell 中执行，服务会立即启动，并在用户登录时自动启动。 ===
powershell -ExecutionPolicy Bypass -File .\scripts\service\Install-Windows.ps1
Get-Content "$env:APPDATA\codex-task\service\token"

# === Windows：仅在当前 PowerShell 要运行 HTTP 请求示例时执行。 ===
$env:CODEX_TASK_TOKEN = (Get-Content "$env:APPDATA\codex-task\service\token" -Raw).Trim()

# === Windows：查看服务状态。 ===
Get-ScheduledTask -TaskName CodexTask | Get-ScheduledTaskInfo

# === Windows：持续查看日志；按 Ctrl+C 退出日志查看，不会停止服务。 ===
Get-Content "$env:APPDATA\codex-task\service\service.log" -Wait

# === Windows：配置变化或升级后重启服务。 ===
Restart-ScheduledTask -TaskName CodexTask

# === Windows：不再需要自动启动服务时才卸载；保留全局 npm 包和 Codex 数据。 ===
powershell -ExecutionPolicy Bypass -File .\scripts\service\Uninstall-Windows.ps1

# === 场景：只在当前电脑临时测试服务；不需要 Token，关闭终端或按 Ctrl+C 后停止。 ===
codex-task serve --host 127.0.0.1 --port 7777

# === 场景：临时让局域网手机访问；生成临时 Token 并在前台运行，关闭终端后停止，每次执行都会覆盖旧 Token。 ===
node -e 'process.stdout.write(require("node:crypto").randomBytes(32).toString("base64url"))' > ./codex-task-token && chmod 600 ./codex-task-token && codex-task serve --host 0.0.0.0 --port 7777 --token-file ./codex-task-token
cat ./codex-task-token

# === 场景：用 macOS/Linux curl 测试已安装的 HTTP 服务；先按所属平台执行上面的 export。 ===
# === 0.0.0.0 只用于服务监听；手机 URL 应把 127.0.0.1 换成电脑的局域网或 VPN IP。 ===
curl -sS http://127.0.0.1:7777/healthz
curl -sS -X POST http://127.0.0.1:7777/v1/text -H "Authorization: Bearer $CODEX_TASK_TOKEN" -H 'Content-Type: application/json' -d '{"prompt":"把这段需求整理成三条要点"}'
curl -sS -X POST http://127.0.0.1:7777/v1/image -H "Authorization: Bearer $CODEX_TASK_TOKEN" -H 'Content-Type: application/json' -d '{"prompt":"生成一份俯拍健身营养餐","quality":"high","count":1}'
curl -sS -X POST http://127.0.0.1:7777/v1/task -H "Authorization: Bearer $CODEX_TASK_TOKEN" -H 'Content-Type: application/json' -d '{"prompt":"实现营养餐详情页并运行测试","workingDirectory":"/absolute/server/path/to/meal-app","sandboxMode":"danger-full-access","networkAccess":true}'
curl -sS -X POST http://127.0.0.1:7777/v1/tasks/替换为taskId/resume -H "Authorization: Bearer $CODEX_TASK_TOKEN" -H 'Content-Type: application/json' -d '{"prompt":"按单人份展示热量"}'
curl -sS http://127.0.0.1:7777/v1/jobs/替换为jobId -H "Authorization: Bearer $CODEX_TASK_TOKEN"
curl -fL http://127.0.0.1:7777/v1/jobs/替换为jobId/artifacts/0 -H "Authorization: Bearer $CODEX_TASK_TOKEN" -o ./artifact.png
