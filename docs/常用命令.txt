# CodexTask 常用命令：每条命令均为可直接复制执行的单行命令。

# === 安装与升级：全局安装后，Agent 和终端都可以直接调用 codex-task。 ===
npm install -g codex-task
codex-task --version
npm install -g codex-task@latest
npm uninstall -g codex-task
npx --yes codex-task@latest --version
npx --yes codex-task@latest doctor

# === Companion Skill：SkillTruck 负责把 CodexTask Skill 安装给其他 Agent；Skill 本身不会安装 CLI。 ===
npm install -g skilltruck
skilltruck install https://github.com/wangyendt/codex-task --global
npm install -g codex-task
codex-task skill path

# === 诊断与清理：检查运行环境、查看帮助并清理过期任务数据。 ===
codex-task doctor
codex-task gc
codex-task --help
codex-task text --help
codex-task image --help
codex-task task --help
codex-task resume --help

# === 文生文（Direct）：适合不需要本地文件或工具的独立文本任务。 ===
codex-task text "把下面的内容整理成简洁的发布公告" --backend direct
codex-task text --prompt-file ./task.md --backend direct
printf '%s' "$PROMPT" | codex-task text --backend direct
codex-task text "分析这个方案的主要风险" --backend direct --model gpt-5.6-sol --reasoning high
codex-task text "生成一份迁移方案" --backend direct --stream

# === 文生图（Direct）：生成图片并保存到指定目录。 ===
codex-task image "一张简洁、精确的机器人关节爆炸图" --backend direct --output ./artifacts
codex-task image "透明背景的产品图标" --backend direct --output ./artifacts --size 1024x1024 --quality high --background transparent
codex-task image "未来主义工具箱概念图" --backend direct --output ./artifacts --count 4 --concurrency 2

# === 图生图（Direct）：使用一至五张本地参考图编辑图片。 ===
codex-task image "保留主体，把背景改成整洁的摄影棚" --backend direct --image ./reference.png --output ./artifacts
codex-task image "使用第一张的主体和第二张的配色" --backend direct --image ./subject.png --image ./style.png --output ./artifacts

# === Workspace Task（SDK）：让 Codex 在指定工作区执行文件和代码任务。 ===
codex-task task "实现该功能并运行相关测试" --backend sdk --cwd /absolute/path/to/repository
codex-task task "按现有项目约定修复这个问题" --backend sdk --cwd /absolute/path/to/repository --no-followup
codex-task task "整理本地代码，不访问网络" --backend sdk --cwd /absolute/path/to/repository --no-network
codex-task task "修改当前项目并测试" --backend sdk --cwd /absolute/path/to/repository --sandbox workspace-write

# === 恢复任务：task 返回 needs_input 和 taskId 后，向同一 Codex thread 补充答案。 ===
codex-task resume "$TASK_ID" "是，必须保持向后兼容"
printf '%s' "这是用户补充的完整要求" | codex-task resume "$TASK_ID"

# === TypeScript SDK：在应用项目中安装 CodexTask SDK。 ===
npm install codex-task

# === 本地开发：安装依赖并运行质量门禁、测试和构建。 ===
npm install
npm run verify
npm test
npm run build
git status --short --branch

# === npm 与 GitHub 发布检查：查看线上版本、provenance 和 Actions 配置。 ===
npm view codex-task version
cd /tmp && npx --yes codex-task@latest --version
gh run list --repo wangyendt/codex-task
gh secret list --repo wangyendt/codex-task
gh variable list --repo wangyendt/codex-task
npm view codex-task version dist-tags dist.attestations --json
